# ============================================================
# CWIC SCRIPTED DIPLOMATIC ACTIONS
#
# Everything here is priced in POLITICAL POWER through the engine's own `cost`
# field. No second currency. The engine greys the button out and writes the
# price into the tooltip on its own, so none of these need a
# custom_trigger_tooltip for the cost.
#
#     0   Recall Volunteers            - de-escalation is always free
#     0   Revoke Military Cooperation  - so is walking away from a base
#    25   Lift Embargo
#    50   Military Cooperation Accord  - transit, ports and airfields
#    50   Revoke Defence Commitment    - breaking your word costs something
#    75   Embargo
#   100   Defence Commitment           - the guarantee AND the volunteer licence
#
# ------------------------------------------------------------
# THREE RULES LEARNED THE HARD WAY. BREAK ANY AND THE ACTION FAILS SILENTLY.
#
# 1. `ROOT = { ... }` DOES NOT EXECUTE inside complete_effect.
#    Not "PREV misbehaves inside it" - the whole block is skipped. Everything
#    ever put inside one here did nothing: send_embargo, give_guarantee, and
#    the Volunteer_Allowance write. Everything at the TOP LEVEL fired.
#    ROOT works fine as a TARGET (`country = ROOT`), just not as a scope.
#    The escape hatch is `every_country = { limit = { tag = ROOT } ... }`,
#    which is one country and costs nothing.
#
# 2. `diplomatic_relation` READS FROM THE SCOPE OUTWARDS.
#    In scope S, `{ country = C relation = military_access }` means
#    "S HAS ACCESS TO C" - not the reverse. Granting access to the sender
#    therefore has to run in the SENDER's scope pointing at the receiver,
#    which is the other reason everything goes through every_country now.
#
# 3. AN EFFECT CANNOT CREATE A DIPLOMATIC ACTION THAT IS HIDDEN.
#    `give_military_access` did nothing while the two diplomatic_relation
#    lines beside it worked, and `is_diplomatic_action_valid_military_access`
#    is `always = no`. `diplomatic_relation` sets the raw relation and slips
#    past the validity check; the dedicated effects do not. Same story for
#    `send_embargo` vs the hidden embargo action.
#
# 4. TRACK STATE IN A VARIABLE, NOT A RELATION TRIGGER.
#    `gives_military_access_to` and `has_guaranteed` gated nothing. The
#    trade_embargo_check toggle is the one mechanism observed working, so
#    every action here uses it: ROOT.<check>^THIS.
#
# Scope: in visible/selectable/complete_effect, ROOT is the SENDER and THIS is
# the RECEIVER.
# ------------------------------------------------------------

scripted_diplomatic_actions = {

	# ---------- ECONOMIC EMBARGO ----------
	# "Embargo" and "Impose Naval Blockade" are the same engine mechanic under
	# two names; `send_embargo` drives both. This is the only button for it.
	opt_embargo_action  = {
		allowed = {}
		visible = {
			check_variable = {
				ROOT.trade_embargo_check^THIS = 0
			}
		}
		selectable = {
			NOT = { has_war_with = ROOT }
			custom_trigger_tooltip = {
				tooltip = Embargo_Action_1_TT
				check_variable = {
					ROOT.trade_embargo_check^THIS = 0
				}
			}
		}
		requires_acceptance = no
		cost = 75
		command_power = 0
		show_acceptance_on_action_button = no
		icon = 1
		on_sent_effect = {}
		complete_effect = {
			set_variable = {
				ROOT.trade_embargo_check^THIS = 1
			}
			every_country = {
				limit = { tag = ROOT }
				# Both, deliberately. send_embargo is the documented effect but
				# may be refused because the embargo action is hidden (rule 3);
				# the raw relation slips past that. Setting an existing relation
				# active twice is harmless. Delete the loser once one is proven.
				send_embargo = PREV
				diplomatic_relation = { country = PREV relation = embargo active = yes }
			}
			custom_effect_tooltip = EMBARGO_IMPOSED_TT
		}
		reject_effect = {}
		can_be_sent = {}
		send_description = embargo_target_send
		reset_receive_effect = {}
		can_be_accepted = { always = no }
		receive_description = embargo_target_receive
		ai_desire = {
			base = 0
		}
	}

	opt_lift_embargo_action = {
		allowed = {}
		visible = {
			check_variable = {
				ROOT.trade_embargo_check^THIS = 1
			}
		}
		selectable = {
			NOT = { has_war_with = ROOT }
			custom_trigger_tooltip = {
				tooltip = Embargo_Action_2_TT
				check_variable = {
					ROOT.trade_embargo_check^THIS = 1
				}
			}
		}
		requires_acceptance = no
		cost = 25
		command_power = 0
		show_acceptance_on_action_button = no
		icon = 1
		on_sent_effect = {}
		complete_effect = {
			set_variable = {
				ROOT.trade_embargo_check^THIS = 0
			}
			every_country = {
				limit = { tag = ROOT }
				break_embargo = PREV
				diplomatic_relation = { country = PREV relation = embargo active = no }
			}
			custom_effect_tooltip = EMBARGO_LIFTED_TT
		}
		reject_effect = {}
		can_be_sent = {}
		send_description = embargo_lifted_target_send
		reset_receive_effect = {}
		can_be_accepted = { always = no }
		receive_description = embargo_lifted_target_receive
		ai_desire = {
			base = 0
		}
	}

	# ---------- MILITARY COOPERATION ACCORD ----------
	# One action instead of three. The six vanilla access requests are hidden;
	# this grants transit, ports and airfields together.
	#
	# All three go through `diplomatic_relation`. `give_military_access` was
	# tried first and did nothing while the two diplomatic_relation lines beside
	# it worked, so the whole accord uses the mechanism that was observed to fire.
	opt_military_cooperation_action = {
		allowed = {}
		visible = {
			NOT = { is_subject_of = ROOT }
			check_variable = { ROOT.mil_coop_check^THIS = 0 }
		}
		selectable = {
			NOT = { has_war_with = ROOT }
			custom_trigger_tooltip = {
				tooltip = MIL_COOP_NO_ACCESS_TT
				check_variable = { ROOT.mil_coop_check^THIS = 0 }
			}
		}
		requires_acceptance = yes
		cost = 50
		command_power = 0
		show_acceptance_on_action_button = yes
		icon = 1
		on_sent_effect = {}
		complete_effect = {
			# Runs in the SENDER's scope pointing at the receiver, so the sender
			# is the one who gains access. Written the other way round first and
			# it granted the exact opposite.
			set_variable = { ROOT.mil_coop_check^THIS = 1 }
			every_country = {
				limit = { tag = ROOT }
				diplomatic_relation = { country = PREV relation = military_access active = yes }
				diplomatic_relation = { country = PREV relation = docking_rights active = yes }
				diplomatic_relation = { country = PREV relation = air_base_access active = yes }
			}
			custom_effect_tooltip = MIL_COOP_GRANTED_TT
		}
		reject_effect = {}
		can_be_sent = {}
		send_description = mil_coop_send
		reset_receive_effect = {}
		can_be_accepted = { always = yes }
		receive_description = mil_coop_receive
		accept_title = mil_coop_accept
		accept_description = mil_coop_accept_desc
		reject_title = mil_coop_reject
		reject_description = mil_coop_reject_desc
		ai_acceptance = {
			mil_coop_acceptance = {
				base = -100
				modifier = {                       # our patron is asking
					sph_is_sphered = yes
					check_variable = { Sphere_Leader = ROOT.id }
					add = 300
				}
				modifier = {                       # same alliance
					is_in_faction_with = ROOT
					add = 250
				}
				modifier = {                       # same bloc, different patron
					sph_is_sphered = yes
					ROOT = { sph_is_sphered = yes }
					check_variable = { Sphere_Leader = ROOT.Sphere_Leader }
					add = 150
				}
				modifier = {                       # already fighting the same people
					ROOT = { has_war = yes }
					is_in_faction_with = ROOT
					add = 100
				}
				modifier = {
					has_opinion = { target = ROOT value > 100 }
					add = 60
				}
				modifier = {                       # we are embargoing them
					check_variable = { trade_embargo_check^ROOT = 1 }
					add = -500
				}
				modifier = {                       # they are embargoing us
					check_variable = { ROOT.trade_embargo_check^THIS = 1 }
					add = -500
				}
			}
		}
		# AI NOTE. `ai_desire` is how the engine decides whether an AI country
		# offers this at all; anything at or below 0 means never. Deliberately
		# narrow - blocs should build a basing network, not spam every country.
		ai_desire = {
			base = -1
			modifier = {
				THIS = { sph_is_sphered = yes }
				THIS = { check_variable = { Sphere_Leader = ROOT.id } }
				add = 30
			}
			modifier = {
				THIS = { sph_is_sphered = yes }
				THIS = { check_variable = { Sphere_Leader = ROOT.id } }
				ROOT = { has_war = yes }
				add = 40
			}
			modifier = {
				THIS = { is_in_faction_with = ROOT }
				ROOT = { is_faction_leader = yes }
				add = 25
			}
		}
	}

	opt_revoke_military_cooperation_action = {
		allowed = {}
		visible = {
			check_variable = { ROOT.mil_coop_check^THIS = 1 }
		}
		selectable = {
			custom_trigger_tooltip = {
				tooltip = MIL_COOP_HAS_ACCESS_TT
				check_variable = { ROOT.mil_coop_check^THIS = 1 }
			}
		}
		requires_acceptance = no
		cost = 0
		command_power = 0
		show_acceptance_on_action_button = no
		icon = 1
		complete_effect = {
			set_variable = { ROOT.mil_coop_check^THIS = 0 }
			every_country = {
				limit = { tag = ROOT }
				diplomatic_relation = { country = PREV relation = military_access active = no }
				diplomatic_relation = { country = PREV relation = docking_rights active = no }
				diplomatic_relation = { country = PREV relation = air_base_access active = no }
			}
			custom_effect_tooltip = MIL_COOP_REVOKED_TT
		}
		can_be_sent = {}
		send_description = mil_coop_revoke_send
		can_be_accepted = { always = no }
		receive_description = mil_coop_revoke_receive
		ai_desire = {
			base = 0
		}
	}

	# ---------- DEFENCE COMMITMENT ----------
	# Deliberately NOT a guarantee. It grants no engine relation at all - it
	# writes ONE variable, the volunteer licence, and that licence also opens
	# lend-lease. Guarantee stayed bundled in at first and behaved badly: it
	# dragged the patron into wars the commitment was specifically meant to keep
	# them out of. The vanilla Guarantee Independence button is still there for
	# anyone who wants that promise; this is the other one.
	opt_defence_commitment_action = {
		allowed = {}
		visible = {
			NOT = { is_subject_of = ROOT }
			check_variable = { ROOT.def_commit_check^THIS = 0 }
		}
		selectable = {
			NOT = { has_war_with = ROOT }
			# Bloc leaders only. This action now speaks for a whole faction -
			# see the propagation in complete_effect - so it cannot be something
			# any member signs on their own.
			#
			# Deliberately in `selectable`, not `visible`: a failed trigger here
			# shows the player a red prerequisite line instead of silently
			# deleting the button, which is how the last three bugs hid.
			custom_trigger_tooltip = {
				tooltip = GUARANTEE_LEADER_ONLY_TT
				ROOT = { is_faction_leader = yes }
			}
			custom_trigger_tooltip = {
				tooltip = DEFENCE_COMMITMENT_NONE_TT
				check_variable = { ROOT.def_commit_check^THIS = 0 }
			}
		}
		requires_acceptance = yes
		cost = 100
		command_power = 0
		show_acceptance_on_action_button = yes
		icon = 1
		complete_effect = {
			set_variable = { ROOT.def_commit_check^THIS = 1 }
			every_country = {
				limit = { tag = ROOT }
				add_ai_strategy = {
					type = send_volunteers_desire
					id = PREV
					value = 200
				}
				# TARGETS ARE `PREV`, NOT AN EVENT TARGET.
				# give_guarantee documents its supported targets as
				# THIS/ROOT/PREV/FROM/OWNER/CONTROLLER/OCCUPIED/CAPITAL - event
				# targets are NOT on that list, so `give_guarantee =
				# event_target:x` silently did nothing while the variable write
				# beside it worked. That is why "Withdraw Guarantee" appeared
				# (def_commit_check was set) with no guarantee behind it.
				#
				# Inside every_country, PREV is the scope outside the loop: the
				# receiving country. Which is exactly what we want to guarantee.
				set_variable = { Volunteer_Allowance^PREV = 1 }
				diplomatic_relation = { country = PREV relation = guarantee active = yes }
				# The actual guarantee. It was taken out when this action was
				# still called a "Defence Commitment" and was not supposed to
				# read as a guarantee at all - but now it IS the guarantee, so
				# the relation has to exist or nothing shows in the diplomacy
				# panel and the button does nothing visible.
				#
				# The reason it was removed - being dragged into the client's
				# war - is handled instead by BREAKING_GUARANTEE_PENALTY = 0 in
				# IC_Defines.lua: the call still arrives, but declining is free.
				give_guarantee = PREV
				# How many we carry. Only the AI reads it - it is what stops a
				# superpower guaranteeing the entire map.
				add_to_variable = { def_commit_count = 1 }
			}
			# THE BLOC FOLLOWS THE LEADER. Same shape as the embargo: the whole
			# faction gets the volunteer and lend-lease channel, not just the
			# signer. NATO guarantees Greece, so every NATO member can send
			# volunteers there.
			#
			# The guarantee RELATION stays with the leader alone on purpose -
			# fifteen members each guaranteeing the same country would mean
			# fifteen call-to-war popups, which is the thing we removed.
			every_country = {
				limit = { is_in_faction_with = ROOT }
				set_variable = { Volunteer_Allowance^PREV = 1 }
				# PERMISSION IS NOT DESIRE.
				# Volunteer_Allowance only lets the button be pressed. The AI
				# decides whether to press it from send_volunteers_desire, and
				# CWIC's existing ai_strategy files are almost all NEGATIVE
				# entries blocking specific pairs - nothing ever told an ally to
				# WANT to reinforce a guaranteed country, so nobody went.
				add_ai_strategy = {
					type = send_volunteers_desire
					id = PREV
					value = 200
				}
			}
			custom_effect_tooltip = DEFENCE_COMMITMENT_LICENCE_TT
		}
		reject_effect = {}
		can_be_sent = {}
		send_description = defence_commitment_send
		reset_receive_effect = {}
		can_be_accepted = { always = yes }
		receive_description = defence_commitment_receive
		accept_title = defence_commitment_accept
		accept_description = defence_commitment_accept_desc
		reject_title = defence_commitment_reject
		reject_description = defence_commitment_reject_desc
		ai_acceptance = {
			defence_commitment_acceptance = {
				base = -50
				modifier = {                       # our own patron
					sph_is_sphered = yes
					check_variable = { Sphere_Leader = ROOT.id }
					add = 300
				}
				modifier = {
					is_in_faction_with = ROOT
					add = 200
				}
				modifier = {                       # same bloc
					sph_is_sphered = yes
					ROOT = { sph_is_sphered = yes }
					check_variable = { Sphere_Leader = ROOT.Sphere_Leader }
					add = 120
				}
				modifier = {                       # we are in trouble
					has_war = yes
					add = 100
				}
				modifier = {
					has_opinion = { target = ROOT value < 0 }
					add = -150
				}
			}
		}
		# AI RESTRAINT. Left alone, a superpower signs one of these with every
		# country it likes and the guarantee stops meaning anything. Three brakes:
		# a deep negative base, a requirement that the target actually be in
		# trouble, and a rising penalty for every commitment already carried - so
		# the tenth costs far more desire than the first.
		ai_desire = {
			base = -40
			modifier = {                           # our own client, under fire
				THIS = { sph_is_sphered = yes }
				THIS = { check_variable = { Sphere_Leader = ROOT.id } }
				THIS = { has_war = yes }
				add = 70
			}
			modifier = {                           # an ally in a war
				THIS = { is_in_faction_with = ROOT }
				THIS = { has_war = yes }
				add = 45
			}
			modifier = {                           # superpowers court the unaligned at war
				ROOT = { OR = { tag = USA tag = SOV } }
				THIS = { sph_is_free = yes  has_war = yes  is_subject = no }
				add = 35
			}
			modifier = {                           # we already carry a few
				ROOT = { check_variable = { def_commit_count > 2 } }
				add = -25
			}
			modifier = {                           # we carry a lot
				ROOT = { check_variable = { def_commit_count > 5 } }
				add = -60
			}
			modifier = {                           # this is a habit now
				ROOT = { check_variable = { def_commit_count > 9 } }
				add = -150
			}
		}
	}

	opt_revoke_defence_commitment_action = {
		allowed = {}
		visible = {
			check_variable = { ROOT.def_commit_check^THIS = 1 }
		}
		selectable = {
			custom_trigger_tooltip = {
				tooltip = DEFENCE_COMMITMENT_HAS_TT
				check_variable = { ROOT.def_commit_check^THIS = 1 }
			}
		}
		requires_acceptance = no
		cost = 50
		command_power = 0
		show_acceptance_on_action_button = no
		icon = 1
		complete_effect = {
			set_variable = { ROOT.def_commit_check^THIS = 0 }
			every_country = {
				limit = { tag = ROOT }
				set_variable = { Volunteer_Allowance^PREV = 0 }
				diplomatic_relation = { country = PREV relation = guarantee active = no }
				subtract_from_variable = { def_commit_count = 1 }
				clamp_variable = { var = def_commit_count min = 0 max = 100 }
			}
			every_country = {
				limit = { is_in_faction_with = ROOT }
				set_variable = { Volunteer_Allowance^PREV = 0 }
				# Cancels the +200 added when the guarantee was signed.
				add_ai_strategy = {
					type = send_volunteers_desire
					id = PREV
					value = -200
				}
			}
			custom_effect_tooltip = DEFENCE_COMMITMENT_REVOKE_TT
		}
		can_be_sent = {}
		send_description = defence_commitment_revoke_send
		can_be_accepted = { always = no }
		receive_description = defence_commitment_revoke_receive
		ai_desire = {
			base = 0
		}
	}

	# NOTE ON TARGETS, DO NOT REGRESS THIS:
	# Effects document which targets they accept - give_guarantee,
	# recall_volunteers_from, send_embargo and break_embargo all list
	# THIS/ROOT/PREV/FROM/OWNER/CONTROLLER/OCCUPIED/CAPITAL and NOT event
	# targets. Passing `event_target:x` to them is a SILENT no-op: the action
	# fires, the variables update, and the actual effect never happens. That is
	# why Recall Volunteers did nothing and the guarantee never appeared.
	# Inside `every_country = { limit = { tag = ROOT } }`, PREV is the receiving
	# country. Use PREV.

	# ---------- SECURE A BLOC ----------
	# You do not lobby fifty countries, you lobby whoever speaks for them.
	# Only BLOC LEADERS can be targeted (has_country_flag = cwic_bloc_leader,
	# authored in CWIC_un_blocs.txt and moved around by content). Everyone
	# following that leader votes with you too - see cwic_un_apply_pledges.
	#
	# Priced at 100 because it is not one vote. The Latin American bloc alone
	# is twenty. Compare: a permanent member burns 250 PP to veto, so buying
	# two or three blocs still costs more than killing a resolution outright -
	# which is right, because a veto only blocks and a bloc also BUILDS.
	opt_un_lobby_action = {
		allowed = {}
		visible = {
			has_country_flag = cwic_bloc_leader
			NOT = { has_variable = un_pledged_to }
		}
		selectable = {
			NOT = { has_war_with = ROOT }
			custom_trigger_tooltip = {
				tooltip = UN_LOBBY_LEADER_TT
				has_country_flag = cwic_bloc_leader
			}
			custom_trigger_tooltip = {
				tooltip = UN_LOBBY_FREE_TT
				NOT = { has_variable = un_pledged_to }
			}
		}
		requires_acceptance = yes
		cost = 100
		command_power = 0
		show_acceptance_on_action_button = yes
		icon = 1
		complete_effect = {
			set_variable = { un_pledged_to = ROOT.id }
			# The flag is what expires. cwic_un_apply_pledges only honours a
			# pledge while it is set, and UN_RESET_VOTE_STATE clears the
			# variable once it lapses.
			set_country_flag = { flag = un_pledged value = 1 days = 730 }
			custom_effect_tooltip = UN_LOBBY_PLEDGED_TT
		}
		reject_effect = {}
		can_be_sent = {}
		send_description = un_lobby_send
		reset_receive_effect = {}
		can_be_accepted = { always = yes }
		receive_description = un_lobby_receive
		accept_title = un_lobby_accept
		accept_description = un_lobby_accept_desc
		reject_title = un_lobby_reject
		reject_description = un_lobby_reject_desc
		ai_acceptance = {
			un_lobby_acceptance = {
				base = -20
				modifier = {                       # our patron asks
					sph_is_sphered = yes
					check_variable = { Sphere_Leader = ROOT.id }
					add = 250
				}
				modifier = {
					is_in_faction_with = ROOT
					add = 200
				}
				modifier = {                       # they guarantee us
					check_variable = { ROOT.Volunteer_Allowance^THIS > 0 }
					add = 150
				}
				modifier = {
					has_opinion = { target = ROOT value > 50 }
					add = 80
				}
				modifier = {
					has_opinion = { target = ROOT value < -25 }
					add = -200
				}
				modifier = {                       # we are embargoing them
					check_variable = { ROOT.trade_embargo_check^THIS = 1 }
					add = -300
				}
			}
		}
		ai_desire = {
			base = -1
			modifier = {                           # bloc leaders work the room
				ROOT = { is_faction_leader = yes }
				THIS = { OR = { has_idea = un_member has_idea = UN_Security_Council_Member_Perm } }
				add = 20
			}
		}
	}

	# ---------- RECALL VOLUNTEERS ----------
	# Free. De-escalation should never be gated behind a price.
	recall_volunteers = {

		allowed = {
			ROOT = { is_ai = no }
		}
		visible = {
			always = yes
		}
		selectable = {
			if = {
				limit = {
					NOT = { has_volunteers_amount_from = { tag = ROOT count > 0 } }
				}
				custom_trigger_tooltip = {
					tooltip = RECALL_VOLUNTEERS_TOOLTIP
					always = no
				}
			}
			else = {
				custom_trigger_tooltip = {
					tooltip = RECALL_VOLUNTEERS_TOOLTIP_G
					always = yes
				}
			}
		}

		cost = 0
		command_power = 0

		show_acceptance_on_action_button = no

		icon = 1

		complete_effect = {
			every_country = {
				limit = { tag = ROOT }
				recall_volunteers_from = PREV
			}
		}

		ai_acceptance = {
			recall_volunteers_acceptance = {
				base = 100
			}
		}
	}
}
