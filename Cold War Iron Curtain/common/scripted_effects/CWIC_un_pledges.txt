# ============================================================
# PLEDGED VOTES - APPLIED WHEN THE PATRON VOTES, NOT BEFORE
#
# WHY THIS EXISTS: the AI tallies its votes when the resolution STARTS
# (UN_START_SCRIPTED_RESOLUTION runs UN_ai_vote_sorting over every member),
# but the player votes during the ten-day window that follows. So the pledge
# term in un_ai_vote_weights could never see the patron's vote flag - it had
# already been read. A bought vote was therefore always ignored.
#
# The fix is to stop asking the AI to predict the patron and instead move the
# pledged countries when the patron actually votes.
#
# cwic_un_apply_pledges runs in the VOTING COUNTRY's scope, from each of the
# four vote buttons in UN_Tab.txt.
#
# BLOCS: the player buys a BLOC LEADER, not a country. Everyone whose
# cwic_bloc_leader_of points at that leader follows too. See CWIC_un_blocs.txt.
# ============================================================

cwic_un_apply_pledges = {
	every_country = {
		limit = {
			OR = {
				# the bloc leader we bought
				AND = {
					has_variable = un_pledged_to
					has_country_flag = un_pledged
					check_variable = { un_pledged_to = ROOT.id }
				}
				# and everyone who follows that leader - this is the whole
				# point of blocs: one purchase, one bloc
				AND = {
					has_variable = cwic_bloc_leader_of
					var:cwic_bloc_leader_of = {
						has_variable = un_pledged_to
						has_country_flag = un_pledged
						check_variable = { un_pledged_to = ROOT.id }
					}
				}
			}
			NOT = { tag = ROOT }
			# A country will not follow us into vetoing itself, and a veto is a
			# permanent member's own prerogative anyway.
			NOT = { check_variable = { UNS.un_nation_target = THIS.id } }
		}
		clr_country_flag = UN_YES_ONGOING
		clr_country_flag = UN_NO_ONGOING
		clr_country_flag = UN_ABSTAIN_ONGOING
		if = {
			limit = { ROOT = { has_country_flag = UN_YES_ONGOING } }
			set_country_flag = UN_YES_ONGOING
		}
		else_if = {
			# A patron's veto still reads as a NO to everyone following them.
			limit = {
				ROOT = {
					OR = {
						has_country_flag = UN_NO_ONGOING
						has_country_flag = UN_VETO_ONGOING
					}
				}
			}
			set_country_flag = UN_NO_ONGOING
		}
		else = {
			set_country_flag = UN_ABSTAIN_ONGOING
		}
	}
	cwic_un_recount = yes
}

# Rebuilds every list and count from the vote flags. The vote buttons keep
# their own running +1/-1 arithmetic, which drifts as soon as anything else
# moves a vote; this is the authority.
cwic_un_recount = {
	clear_array = UNS.un_vote_yes_list
	clear_array = UNS.un_vote_no_list
	clear_array = UNS.un_vote_abstain_list
	clear_array = UNS.un_vote_veto_list
	set_variable = { UNS.un_yes_count = 0 }
	set_variable = { UNS.un_no_count = 0 }
	set_variable = { UNS.un_abstain_count = 0 }
	set_variable = { UNS.un_veto_count = 0 }
	every_country = {
		limit = {
			OR = {
				has_idea = un_member
				has_idea = UN_Security_Council_Member_Perm
			}
		}
		if = {
			limit = { has_country_flag = UN_VETO_ONGOING }
			add_to_array = { UNS.un_vote_veto_list = THIS.id }
			add_to_variable = { UNS.un_veto_count = 1 }
			add_to_variable = { UNS.un_no_count = 1 }
		}
		else_if = {
			limit = { has_country_flag = UN_YES_ONGOING }
			add_to_array = { UNS.un_vote_yes_list = THIS.id }
			add_to_variable = { UNS.un_yes_count = 1 }
		}
		else_if = {
			limit = { has_country_flag = UN_NO_ONGOING }
			add_to_array = { UNS.un_vote_no_list = THIS.id }
			add_to_variable = { UNS.un_no_count = 1 }
		}
		else_if = {
			limit = { has_country_flag = UN_ABSTAIN_ONGOING }
			add_to_array = { UNS.un_vote_abstain_list = THIS.id }
			add_to_variable = { UNS.un_abstain_count = 1 }
		}
	}
}
