MLA_clamp_emergency_momentum = {
	clamp_variable = {
		var = emergency_momentum
		min = -250
		max = 250
	}
}

MLA_register_emergency_operation_win = {
	set_temp_variable = { emergency_streak_shift = MLA_emergency_win_streak }
	multiply_temp_variable = { emergency_streak_shift = 5 }	# +5 momentum per consecutive prior win
	clamp_temp_variable = { var = emergency_streak_shift min = 0 max = 20 }
	add_to_variable = { emergency_momentum = emergency_momentum_base }
	add_to_variable = { emergency_momentum = emergency_streak_shift }
	add_to_variable = { MLA_emergency_win_streak = 1 }
	set_variable = { MLA_emergency_lose_streak = 0 }
	add_to_variable = { MLA_operations_won = 1 }	# scales the size of the General Uprising package
	MLA_clamp_emergency_momentum = yes
}

MLA_register_emergency_operation_loss = {
	set_temp_variable = { emergency_streak_shift = MLA_emergency_lose_streak }
	multiply_temp_variable = { emergency_streak_shift = 8 }	# -8 momentum per consecutive prior loss: defeats snowball harder than victories
	clamp_temp_variable = { var = emergency_streak_shift min = 0 max = 32 }
	subtract_from_variable = { emergency_momentum = emergency_momentum_base }
	subtract_from_variable = { emergency_momentum = emergency_streak_shift }
	add_to_variable = { MLA_emergency_lose_streak = 1 }
	set_variable = { MLA_emergency_win_streak = 0 }
	MLA_clamp_emergency_momentum = yes
}

MLA_shift_emergency_momentum = {	# MLA scope; expects temp emergency_momentum_shift, signed. Diplomatic swings deliberately leave the win/lose streaks alone
	add_to_variable = { emergency_momentum = emergency_momentum_shift }
	MLA_clamp_emergency_momentum = yes
}
