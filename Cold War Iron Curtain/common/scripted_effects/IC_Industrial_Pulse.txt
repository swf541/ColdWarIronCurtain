		set_variable = { food_penalty = food_deficit }
		divide_variable = { food_penalty = food_consumption }
	}

	# Keep the resource and crisis calculations on the full shortage ratio, but
	# scale and cap the direct stability/war-support hit at 10%.
	set_variable = { food_penalty_political = food_penalty }
	multiply_variable = { food_penalty_political = -0.1 }
	clamp_variable = {
		var = food_penalty_political
		min = -0.1
		max = 0
	}

	clr_country_flag = food_famine_extreme
	clr_country_flag = food_famine
	clr_country_flag = food_shortages
	clr_country_flag = food_highprices

	if = {
		limit = {
			check_variable = { food_penalty > 3 }
		}
		set_country_flag = food_famine_extreme
	}
	else_if = {
		limit = {
			check_variable = { food_penalty > 0.8 }
		}
		set_country_flag = food_famine
	}
	else_if = {
		limit = {
			check_variable = { food_penalty > 0.2 }
		}
		set_country_flag = food_shortages
	}
	else_if = {
		limit = {
			check_variable = { food_penalty > 0 }
		}
		set_country_flag = food_highprices
	}
	else = { }

	# Force refresh of famine warning display
	if = {
		limit = {
			check_variable = { food_penalty > 0 }
		}

		set_temp_variable = { show = 1 }
		subtract_from_temp_variable = { show = show_food_main }
		set_variable = { show_food_main = show }
	}

	country_event = { id = food_pulse.1 days = 1 } # call recalc in 1 day (because of added capital bonus which only reflects after daily tick)

}

industrial_pulse_food_script = {
	set_variable = { industrial_pulse_food_script_check = 1 }
	calc_pollution_country_factors = yes
	every_owned_state = {
		calc_state_pollution_effect = yes
		food_consumption_calculator = yes
		add_to_variable = { PREV.fossil_fuel_plant_quantity = non_damaged_building_level@fossil_fuel_powerplant }

		add_to_variable = { PREV.dockyard_quantity = non_damaged_building_level@dockyard }

		add_to_variable = { PREV.steel_factory_quantity = non_damaged_building_level@steel_factory }
		add_to_variable = { PREV.synthetic_refinery_quantity = non_damaged_building_level@synthetic_refinery }

		if = { limit = { has_resources_amount = { resource = rubber amount > 0 delivered = yes } } # counts number of Agri-Complexes that actually deliver
			add_to_variable = { PREV.agri_industrial_complex_quantity = non_damaged_building_level@agri_industrial_complex }
			#if = { limit = { is_owned_by = AST } log = "AST: [THIS.GetName] - agri complexes: [?temp_agri_industrial_complex_quantity]" }		
		}
		if = {
			limit = { has_resources_amount = { resource = rubber amount > 0 }
				is_capital = no # check for leftover food in all states except capital
			}
			set_temp_variable = { temp_agri_industrial_complex_quantity = non_damaged_building_level@agri_industrial_complex } # number of Agri-Complexes in State
			set_temp_variable = { temp_food_produced = resource@rubber } # total food production in State
			subtract_from_temp_variable = { temp_food_produced = temp_agri_industrial_complex_quantity } # deduct food from Agri-Complexes
			if = {
				limit = { NOT = { check_variable = { temp_food_produced = 0 } } }
				#log = "	[ROOT.GetTag]: removed/added [?temp_food_produced] from [THIS.GetName]"
				multiply_temp_variable = { temp_food_produced = -1 }
				add_resource = {
					type = rubber
					amount = temp_food_produced # remove/add leftovers from State (can happen in some cases where a country loses capital to another)
				}
			}
		}
		#	add_to_variable = { PREV.foodCheck = resource@rubber }
		#Farm Production
		set_variable = { farm_output = 1 }

		# Biome-Based Modifiers
		if = { limit = { has_state_flag = desert } add_to_variable = { farm_output = -0.5 } }
		else_if = { limit = { has_state_flag = desertlush } add_to_variable = { farm_output = 1.0 } }
		else_if = { limit = { has_state_flag = tundra } add_to_variable = { farm_output = -0.5 } }
		else_if = { limit = { has_state_flag = mountains } add_to_variable = { farm_output = -0.25 } }
		else_if = { limit = { has_state_flag = arid_mountains } add_to_variable = { farm_output = -0.75 } }
		else_if = { limit = { has_state_flag = boreal } add_to_variable = { farm_output = 1.5 } }
		else_if = { limit = { has_state_flag = temperate } add_to_variable = { farm_output = 2.5 } }
		else_if = { limit = { has_state_flag = tropical } add_to_variable = { farm_output = 3.0 } }
		
		else_if = { limit = { has_state_flag = savanna } add_to_variable = { farm_output = 2.0 } }
		else_if = { limit = { has_state_flag = subtropical_monsoon } add_to_variable = { farm_output = 2.5 } }
		else_if = { limit = { has_state_flag = delta } add_to_variable = { farm_output = 3.5 } }
		else_if = { limit = { has_state_flag = wetlands } add_to_variable = { farm_output = 2.5 } }
		else_if = { limit = { has_state_flag = mediterranean } add_to_variable = { farm_output = 2.0 } }
		else_if = { limit = { has_state_flag = steppe } multiply_variable = { farm_output = 1.05 } # Slight productivity bonus }
		else = { }

		# Water Availability Bonuses
		if = { limit = { has_state_flag = basic } multiply_variable = { farm_output = 0.75 } }
		else_if = { limit = { has_state_flag = lakes } multiply_variable = { farm_output = 1.25 } }
		else_if = { limit = { has_state_flag = river } multiply_variable = { farm_output = 1.5 } }
		else_if = { limit = { has_state_flag = riverlarge } multiply_variable = { farm_output = 1.75 } }
		else_if = { limit = { has_state_flag = freshwatersea } multiply_variable = { farm_output = 2.0 } }
		else = { }

		# Farm Tech & Infrastructure
		set_variable = { farm_tech_local = PREV.farm_tech }
		if = { limit = { water_infrastructure < 1 } multiply_variable = { farm_output = 0.5 } }
		else_if = { limit = { water_infrastructure > 1 water_infrastructure < 3 } add_to_variable = { farm_tech_local = 0.25 } }
		else_if = { limit = { water_infrastructure > 2 water_infrastructure < 4 } add_to_variable = { farm_tech_local = 0.5 } }
		else_if = { limit = { water_infrastructure > 3 } add_to_variable = { farm_tech_local = 0.75 } }
		else = { }

		multiply_variable = { farm_output = farm_tech_local }
		multiply_variable = { farm_output = PREV.farm_output_modif }
		add_to_variable = { farm_output = global.eco_output_rate }

		# Building and Efficiency Modifiers
		multiply_variable = { farm_output = non_damaged_building_level@agri_industrial_complex }
		set_variable = { farm_output_efficiency = farm_output }
		multiply_variable = { farm_output_efficiency = resourceEfficiencyInfra }
		add_to_variable = { farm_output = farm_output_efficiency }

		# Modifier Hook
		set_variable = { agri_prod_modifier = PREV.modifier@agricultural_complex_production_bonus }
		add_to_variable = { agri_prod_modifier = 1 }
		multiply_variable = { farm_output = agri_prod_modifier }

		# Pollution Penalty (scaled linearly from 0–600)
		# Land pollution reduces productivity more harshly than water

		set_variable = { land_pollution_penalty = statelandpollution }
		set_variable = { water_pollution_penalty = statewaterpollution }

		set_variable = { land_pollution_penalty_test = land_pollution_penalty }
		# Scale penalties
		divide_variable = { land_pollution_penalty = 600 }
		divide_variable = { water_pollution_penalty = 600 }

		set_variable = { land_pollution_penalty_test2 = land_pollution_penalty }

		# Max 40% land penalty, 20% water penalty
		multiply_variable = { land_pollution_penalty = 0.4 }
		multiply_variable = { water_pollution_penalty = 0.2 }

		# --- Pollution Penalty ---
		# total_pollution_penalty = land_penalty + water_penalty

		set_variable = { total_pollution_penalty = land_pollution_penalty }
		add_to_variable = { total_pollution_penalty = water_pollution_penalty }

		# Convert penalty to multiplier: (1 - penalty)
		set_variable = { pollution_multiplier = 1 }
		subtract_from_variable = { pollution_multiplier = total_pollution_penalty }

		# Apply to farm_output
		multiply_variable = { farm_output = pollution_multiplier }
		if = {  #Colonial or Rentier Adjustment to Food v Cash Crop
			limit = { 
				OR = {
					check_variable = { economic_x_axis = 26 } 
					check_variable = { economic_x_axis = 24 } 
				}
			} 
			multiply_temp_variable = { cropBoost = 0.5 }
		}

		# Final Application
		add_to_variable = { PREV.food_bonus = farm_output }
		
	}
	
	#Food Consumption End
