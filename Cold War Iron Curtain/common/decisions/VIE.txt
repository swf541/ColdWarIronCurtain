	State_Industrialization_Ready = {
		state_target = yes
		target_trigger = {
			FROM = {
				has_state_flag = Not_Industrialized
			}
		}
		cancel_trigger = {
			FROM = {
				NOT = {
					is_owned_and_controlled_by = VIE
				}
			}
		}
		target_array = VIE.owned_controlled_states
		on_map_mode = decision_view_only
		days_remove = 90
		fixed_random_seed = no
		custom_cost_trigger = {
			has_political_power > 24
			check_variable = {
				currentMoney > 0.49
			}
		}
		custom_cost_text = VIE_has_500_mil
		complete_effect = {
			hidden_effect = {
				add_political_power = -25
				add_to_variable = {
					currentMoney = -0.5
				}
			}
		}
		remove_effect = {
			FROM = {
				clr_state_flag = Not_Industrialized
				hidden_effect = {
					random_list = {
						10 = {
							modifier = {
								factor = 1.5
								water_infrastructure > 0
							}
							add_building_construction = {
								type = agri_industrial_complex
								level = 1
								instant_build = yes
							}
						}
						10 = {
							add_building_construction = {
								type = industrial_complex
								level = 1
								instant_build = yes
							}
						}
						10 = {
							add_building_construction = {
								type = steel_factory
								level = 1
								instant_build = yes
							}
						}
					}
