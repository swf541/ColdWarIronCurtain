# Sphere flavour triggers. The influence system is gone; these just read the
# variables content sets. A country "leads a sphere" if it has a member list.
is_sphere_capable = {
	# sphere_tag_list is an ARRAY — has_variable does not detect arrays and
	# returns false for everyone, which un-hides the vanilla access actions.
	check_variable = { sphere_tag_list^num > 0 }
}
sph_is_sphered = {
	has_variable = Sphere_Leader
	NOT = { check_variable = { Sphere_Leader = 0 } }
}
sph_is_free = {
	OR = {
		NOT = { has_variable = Sphere_Leader }
		check_variable = { Sphere_Leader = 0 }
	}
}
