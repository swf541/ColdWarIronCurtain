#	Triggers to control if diplomatic actions are enabled:
#		Format:
#
#		<diplomatic action tag>_enabled_trigger = {
#			<trigger contents>
#		}
#
#		ROOT is the country initiating the diplomatic action.
#		FROM is the recipient or target of the diplomatic action.
#		Note:  For cleaner tooltips, it is recommended that you use
#			if triggers within the enable triggers, and a single
#			custom_trigger_tooltip trigger within each if trigger.
#
#		Example: (Germany is prevented from declaring war on the Soviet Union.)
#
#		DIPLOMACY_WAR_ENABLE_TRIGGER = {
#			if = {
#				limit = {
#					tag = GER
#				}
#				custom_trigger_tooltip = {
#					tooltip = PREVENT_GER_WAR_ON_SOV
#					FROM = {
#						NOT = {
#							tag = SOV
#						}
#					}
#				}
#			}
#		}
# This specifically overrides the rule check as to whether or not you can guarantee other ideologies

			}
		}

		}
	}

DIPLOMACY_LEND_LEASE_ENABLE_TRIGGER = {
	# FREE FOR EVERYONE. The old gate was "same Sphere_Leader", which since the
	# influence system was removed meant 0 = 0 for every untouched country: it
	# let every unsphered pair through and blocked every sphered one. Exactly
	# backwards, which is why lend-lease read as broken.
	#
	# Rather than write a better gate, there is no gate. Lend-lease is how
	# equipment moves in this war and the game rules already cover the cases a
	# host wants to restrict.
	#
	# NOTE: the vanilla lend-lease action CANNOT be made to cost Political
	# Power. There is no LEND_LEASE_COST define and no on_action fires when a
	# contract is signed. The only way to price it is to replace it with a
	# scripted diplomatic action that pays PP and calls send_equipment - a
	# one-shot arms shipment, not a standing contract.
}
