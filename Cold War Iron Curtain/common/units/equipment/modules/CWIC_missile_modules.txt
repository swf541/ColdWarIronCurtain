# ============================================================
# MISSILE MODULES - ADDITIVE, THE APC WAY
#
# These declare VANILLA categories, so they sit in the designer next to
# vanilla's own modules rather than replacing them. Three only: the pieces a
# missile needs that no bomber module covers. Everything else - bombs, jets,
# radar, radio navigation - already exists and the airframe simply allows it.
#
# An earlier version of this file invented four categories of its own
# (cwic_missile_warhead etc). That is what left the air designer with nothing
# it recognised. Do not reintroduce them.
#
# Stats are a first pass, tuned to read clearly in the designer, not balanced.
# ============================================================

equipment_modules = {

	# A warhead is a bomb that arrives on its own. Sits in strat_weapon, so it
	# competes directly with vanilla's bomb bays in the same slot.
	cwic_missile_warhead = {
		abbreviation = "mwh"
		category = strat_weapon
		sfx = sfx_ui_sd_module_armor
		add_stats = {
			air_bombing = 300
			build_cost_ic = 14
		}
		multiply_stats = {
			air_range = -0.20
		}
		build_cost_resources = { steel = 2 }
	}

	# Storable, quick to launch, short-legged. The counterpart to vanilla's
	# liquid rocket_engine_x, which is already in this category.
	cwic_solid_rocket_motor = {
		abbreviation = "srm"
		category = plane_rocket_engine_type
		sfx = sfx_ui_sd_module_engine
		add_stats = {
			thrust = 28
			build_cost_ic = 16
		}
		multiply_stats = {
			air_range = -0.45
			maximum_speed = 0.9
		}
		# Solid fuel IS the reliable one - it stores in a tube for years and
		# has nothing to pump. The bonus is intentional and belongs here where
		# it reads as a bonus rather than buried in multiply_stats.
		add_stats = {
			reliability = 0.08
		}
		build_cost_resources = { tungsten = 2 }
	}

	# Inertial guidance: no operator, no radio to jam, and it does not care
	# about weather. Cheaper and less accurate than vanilla's radar packages.
	cwic_inertial_guidance = {
		abbreviation = "ing"
		category = plane_special_module_radio_navigation
		sfx = sfx_ui_sd_module_electronics
		add_stats = {
			reliability = 0.10
			air_range = 50
			build_cost_ic = 8
		}
		build_cost_resources = { tungsten = 1 }
	}

	# The second warhead step. Everything the first one is, more so - and it
	# eats the range to pay for it, which is the tradeoff the whole airframe
	# is built around.
	cwic_missile_heavy_warhead_module = {
		abbreviation = "mhw"
		category = strat_weapon
		sfx = sfx_ui_sd_module_armor
		add_stats = {
			air_bombing = 600
			build_cost_ic = 30
		}
		multiply_stats = {
			air_range = -0.40
			maximum_speed = -0.10
		}
		build_cost_resources = { steel = 4 }
	}

	# ---- second guidance step ----
	# Terrain matching. Reads the ground against a stored map, which is the
	# thing that turned a missile from an area weapon into an aimed one.
	cwic_terrain_guidance = {
		abbreviation = "trg"
		category = plane_special_module_air_ground_radar
		sfx = sfx_ui_sd_module_electronics
		add_stats = {
			reliability = 0.15
			air_bombing = 80
			build_cost_ic = 22
		}
		build_cost_resources = { tungsten = 3 }
	}

	# ---- second propulsion step ----
	# Staging. The only way to reach intercontinental range, and it costs a
	# whole missile's worth of engine to get there.
	cwic_staged_booster = {
		abbreviation = "stg"
		category = plane_rocket_engine_type
		sfx = sfx_ui_sd_module_engine
		add_stats = {
			thrust = 45
			build_cost_ic = 40
		}
		multiply_stats = {
			air_range = 1.4
			maximum_speed = 0.4
		}
		build_cost_resources = { tungsten = 3  aluminium = 2 }
	}

	# ---- airframe line ----
	cwic_light_airframe = {
		abbreviation = "lgt"
		category = plane_special_module_small
		sfx = sfx_ui_sd_module_armor
		add_stats = { build_cost_ic = 4 }
		multiply_stats = {
			air_range = 0.20
			maximum_speed = 0.15
			air_defence = -0.20
		}
	}
	cwic_hardened_airframe = {
		abbreviation = "hrd"
		category = plane_special_module_medium
		sfx = sfx_ui_sd_module_armor
		add_stats = {
			air_defence = 30
			reliability = 0.10
			build_cost_ic = 12
		}
		multiply_stats = { air_range = -0.15 }
		build_cost_resources = { steel = 2 }
	}
}
