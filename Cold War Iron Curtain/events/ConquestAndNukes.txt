# Coalition Forces Approach the Yalu River
news_event = {
	id = conquest.106 
 immediate = { }
	title = conquest.106.t
	desc = conquest.106.d
	picture = GFX_china_crosses_yalu
	major = yes

	trigger = {
		NOT = { has_global_flag = korean_war_china_can_intervene }
		has_global_flag = CCW_Over
		has_global_flag = PRC_Victory
		has_idea = korean_war
		OR = {
			AND = {
				745 = { is_controlled_by = KOR }
				745 = { is_owned_by = KPA }
			}
			AND = {
				747 = { is_controlled_by = KOR }
				747 = { is_owned_by = KPA }
			}
			AND = {
				1205 = { is_controlled_by = KOR }
				1205 = { is_owned_by = KPA }
			}
			AND = {
				527 = { is_controlled_by = KOR }
				527 = { is_owned_by = KPA }
			}
			KOR = { num_occupied_states > 4 }
		}
	}

	mean_time_to_happen = {
		days = 2
	}

	immediate = {
		set_global_flag = korean_war_china_can_intervene
		set_variable = {
			PRC.Volunteer_Allowance^KPA = 20
		}
		hidden_effect = {
			PRC = {
				declare_war_on = { target = KOR type = annex_everything }
			}
			KPA = {
				add_to_faction = PRC
			}
		}
	}

	option = {
		name = conquest.106.1.a
		trigger = {
			OR = {
				TAG = USA
				TAG = KOR
			}
		}
	}
	option = {
		name = conquest.106.1.b
		trigger = {
			NOT = {
				TAG = KPA
				TAG = USA
				TAG = PRC
			}
		}
	}
	option = {
		name = conquest.106.1.c
		trigger = {
			TAG = KPA
		}
	}
	option = {
		name = conquest.106.1.d
		trigger = {
			TAG = PRC
		}
	}
}
