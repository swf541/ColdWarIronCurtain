# Fire it with:

country_event = {
	id = cwic_sphere.5
	title = cwic_sphere.5.t
	desc = cwic_sphere.5.d
	picture = GFX_report_event_generic_rally

	is_triggered_only = yes

	# Put our thumb on the scale — costs 25 of the influence we built here.
	option = {
		name = cwic_sphere.5.a
		event_target:sph_patron_subject = {
			set_temp_variable = { sph_from = ROOT.id }
			set_temp_variable = { sph_amount = -25 }
			sph_grant_influence = yes
		}
		custom_effect_tooltip = cwic_sphere.5.a.tt
	}

	# Stay out of it.
	option = {
		name = cwic_sphere.5.b
	}
}
