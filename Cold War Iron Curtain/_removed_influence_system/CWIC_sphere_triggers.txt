# ============================================================
# CWIC Sphere of Influence — scripted triggers (content-creator API)
#
# QUICK REFERENCE — things you can check directly, no scripted trigger needed:
#   - Who owns their sphere:   check_variable = { Sphere_Leader = USA.id }   (target scope)
#   - Economic accord w/ USA:  has_country_flag = sph_econ_accord_USA        (target scope)
#   - Military accord w/ USA:  has_country_flag = sph_mil_accord_USA         (target scope)
#   - Accord ideas:            has_idea = sph_econ_accord_idea / sph_mil_accord_idea / sphere_member_trade
#   - Is X in my sphere:       is_in_array = { sph_members = X }             (leader scope)
#   - Am I a sphere power:     is_sphere_capable = yes
#
# For an influence-LEVEL check (e.g. "USA has >50 influence here"), first run the
# effect sph_ce_store_influence in an effect block (ROOT = the leader), then:
#   check_variable = { sph_query_influence > 50 }
# ============================================================

# A country is sphere-capable (can influence others) if it owns a non-empty sphere_tag_list.
is_sphere_capable = {
	check_variable = { sphere_tag_list^num > 0 }
}

# Is this country in ANY sphere? (target scope)
sph_is_sphered = {
	has_variable = Sphere_Leader
	NOT = { check_variable = { Sphere_Leader = 0 } }
}

# Is this country free of any sphere? (target scope)
sph_is_free = {
	OR = {
		NOT = { has_variable = Sphere_Leader }
		check_variable = { Sphere_Leader = 0 }
	}
}

# Target scope: is ROOT my sphere leader? (focus available/bypass from the member side)
sph_root_is_my_leader = {
	has_variable = Sphere_Leader
	check_variable = { Sphere_Leader = ROOT.id }
}

# Target scope: is FROM my sphere leader?
sph_from_is_my_leader = {
	has_variable = Sphere_Leader
	check_variable = { Sphere_Leader = FROM.id }
}

# GP scope: do I have any sphere members?
sph_has_members = {
	check_variable = { sph_members^num > 0 }
}

# GP (leader) scope: is FROM one of my sphere members?
sph_from_is_my_member = {
	is_in_array = { sph_members = FROM.id }
}

# Target scope: is ROOT currently influencing me? (ROOT = a sphere power)
sph_root_is_influencing_me = {
	is_in_array = { ROOT.sph_active_ids = THIS.id }
}

# AI helper: is THIS country a valid influence target for the courting GP
# (saved as event_target:sph_ai_me by sph_ai_monthly)?
sph_ai_eligible_target = {
	NOT = { is_sphere_capable = yes }
	NOT = { tag = event_target:sph_ai_me }
	NOT = { is_subject = yes }
	OR = {
		NOT = { has_variable = Sphere_Leader }
		check_variable = { Sphere_Leader = 0 }
	}
	OR = {
		NOT = { is_in_faction = yes }
		is_in_faction_with = event_target:sph_ai_me
	}
	NOT = { has_war_with = event_target:sph_ai_me }
	NOT = { is_in_array = { event_target:sph_ai_me.sph_active_ids = THIS.id } }
}

# =============================================================
# IDEOLOGICAL COMPATIBILITY
# -------------------------------------------------------------
# THE POLICY LIVES HERE. Two buckets, and accords are refused across the
# Iron Curtain between them. Everything else (nationalists, monarchies,
# islamists, non-aligned) can deal with either side — Washington armed the
# Shah and Moscow armed Nasser, so those pairings must stay legal.
# To change the rules, edit these two lists and nothing else.
# =============================================================
sph_bloc_left = {
	OR = {
		has_government = communism
		has_government = maoism
		has_government = trotskyism
		has_government = socialist
	}
}
sph_bloc_west = {
	OR = {
		has_government = democratic
		has_government = liberal
		has_government = centrist
		has_government = conservative
	}
}

# Escape hatch for content: Tito, Nasser, Ceausescu, anyone playing both sides.
# Set sph_ignore_ideology on EITHER country and the curtain stops applying.
# TARGET scope, ROOT = the great power (use in diplomatic actions / GUI).
sph_curtain_blocked_root = {
	OR = {
		AND = { sph_bloc_left = yes  ROOT = { sph_bloc_west = yes } }
		AND = { sph_bloc_west = yes  ROOT = { sph_bloc_left = yes } }
	}
	NOT = { has_country_flag = sph_ignore_ideology }
	NOT = { ROOT = { has_country_flag = sph_ignore_ideology } }
}
# Same test, TARGET scope with temp sph_from = leader id (use inside effects).
sph_curtain_blocked_from = {
	OR = {
		AND = { sph_bloc_left = yes  var:sph_from = { sph_bloc_west = yes } }
		AND = { sph_bloc_west = yes  var:sph_from = { sph_bloc_left = yes } }
	}
	NOT = { has_country_flag = sph_ignore_ideology }
	NOT = { var:sph_from = { has_country_flag = sph_ignore_ideology } }
}

# Economic accord: blocked only across the curtain.
sph_can_econ_accord_root = { NOT = { sph_curtain_blocked_root = yes } }

# Military accord: stricter. Across the curtain is out, and you do not get
# basing rights in a country that already answers to a rival power.
sph_can_mil_accord_root = {
	NOT = { sph_curtain_blocked_root = yes }
	OR = {
		sph_is_free = yes
		check_variable = { Sphere_Leader = ROOT.id }
	}
}
