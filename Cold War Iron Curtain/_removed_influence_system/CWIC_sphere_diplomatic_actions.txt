# ============================================================
# CWIC Sphere of Influence — native diplomatic actions (accord model)
# ROOT = sender (Great Power) ; THIS = target country.

scripted_diplomatic_actions = {

	# Start influencing (uses one influence slot)
	opt_sphere_influence = {
		allowed = {}
		visible = {
			ROOT = { is_sphere_capable = yes }
			NOT = { is_sphere_capable = yes }
			NOT = { check_variable = { sph_disp_active = 1 } }
			NOT = { check_variable = { Sphere_Leader = ROOT } }
		}
		selectable = {
			NOT = { has_war_with = ROOT }
			custom_trigger_tooltip = {
				tooltip = sph_dip_need_slot_TT
				ROOT = { check_variable = { sph_active_ids^num < sph_slots } }
			}
		}
		requires_acceptance = no
		cost = 0
		command_power = 0
		show_acceptance_on_action_button = no
		icon = 1
		on_sent_effect = {}
		complete_effect = {
			set_temp_variable = { sph_tid2 = THIS.id }
			ROOT = { set_temp_variable = { sph_ptarget = sph_tid2 } sph_toggle_influence = yes }
		}
		reject_effect = {}
		can_be_sent = {}
		send_description = sph_dip_influence_send
		reset_receive_effect = {}
		can_be_accepted = { always = no }
		receive_description = sph_dip_generic_receive
		ai_desire = { base = 0 }
	}

	opt_sphere_econ_accord = {
		allowed = {}
		visible = {
			# hard gates: not in another power's sphere, not in a rival faction, not a foreign subject
			OR = {

			}

				NOT = { is_in_faction = yes }

			NOT = { is_sphere_capable = yes }
			check_variable = { sph_disp_stage = 0 }

		}
		selectable = {
			NOT = { has_war_with = ROOT }
			custom_trigger_tooltip = {
				tooltip = sph_tt_econ_ideology
				sph_can_econ_accord_root = yes
			}
		}
		requires_acceptance = no
		cost = 25
		command_power = 0
		show_acceptance_on_action_button = no
		icon = 1
		on_sent_effect = {}
		complete_effect = {
			set_temp_variable = { sph_from = ROOT.id }
			THIS = { sph_sign_econ_accord = yes }
		}
		reject_effect = {}
		can_be_sent = {}
