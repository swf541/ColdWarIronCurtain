# ============================================================
# CWIC Sphere of Influence — on_actions (accord model)
# 2026-07-20
# ============================================================
on_actions = {

	# ---- one-time migration: existing sphere memberships -> new model; influence starts at 0 ----
	on_startup = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = sph_migrated } }
				set_global_flag = sph_migrated

				every_country = { sph_init_country = yes }

				# Countries already in a sphere_tag_list become full members
				# (Sphere_Leader + both accords implied at stage 2).
				every_country = {
					limit = { is_sphere_capable = yes }
					set_variable = { sph_leader_id = THIS.id }
					for_each_loop = {
						array = sphere_tag_list
						value = memberid
						if = {
							limit = {
								NOT = { check_variable = { memberid = sph_leader_id } }
								country_exists = var:memberid
							}
							if = {
								limit = { NOT = { is_in_array = { sph_members = memberid } } }
								add_to_array = { sph_members = memberid }
							}
							var:memberid = {
								if = { limit = { OR = { NOT = { has_variable = Sphere_Leader } check_variable = { Sphere_Leader = 0 } } }
									set_variable = { Sphere_Leader = PREV.sph_leader_id }
									add_ideas = sphere_member_trade
									set_temp_variable = { sph_from = PREV.sph_leader_id }
									sph_ensure_pair = yes
									add_to_variable = { sph_gp_stage^sph_idx = 2 }   # accords implied
									var:sph_from = { save_event_target_as = sph_mig_leader }
									add_opinion_modifier = { target = event_target:sph_mig_leader modifier = sph_sphere_member }
								}
							}
						}
					}
				}
				# Old influence_array values are intentionally NOT imported — influence
				# starts at 0 everywhere so progress only reflects player actions.
			}
		}
	}

	# ---- weekly accrual ----
	on_weekly = {
		effect = {
			if = {
				limit = { is_sphere_capable = yes }
				sph_weekly_tick = yes
			}
			sph_monthly_validate = yes
		}
	}

	# ---- monthly: AI actors, leader economics, embargo sync ----
	on_monthly = {
		effect = {
			if = {
				limit = { is_sphere_capable = yes }
				# AI Great Powers play the influence game
				if = {
					limit = { is_ai = yes }
					sph_ai_monthly = yes
				}
				# leader economics: flat monthly income per sphere member
				if = {
					limit = { check_variable = { sph_members^num > 0 } }
					set_temp_variable = { sph_income = sph_members^num }
					multiply_temp_variable = { sph_income = 0.01 }
					add_to_variable = { currentMoney = sph_income }
				}
				# Embargo sync, both directions: members adopt the leader's embargoes and
				# drop them again when the leader lifts one. sph_forced_emb marks the ones the
				# sphere imposed, so a member's own embargoes are never touched.
				# ponytail: O(countries x members) once a month per leader, same cost as before;
				# make it event-driven if the monthly tick ever shows up in a profile.
				every_country = {
					limit = {
						NOT = { tag = PREV }
						NOT = { is_in_array = { PREV.sph_members = THIS.id } }
					}
					save_event_target_as = sph_emb_target
					set_temp_variable = { sph_emb_on = 0 }
					if = {
						limit = { check_variable = { PREV.trade_embargo_check^THIS = 1 } }
						set_temp_variable = { sph_emb_on = 1 }
					}
					PREV = {
						for_each_loop = {
							array = sph_members
							value = mid
							if = {
								limit = { country_exists = var:mid }
								var:mid = {
									if = {
										limit = {
											check_variable = { sph_emb_on = 1 }
											NOT = { check_variable = { sph_forced_emb^event_target:sph_emb_target = 1 } }
										}
										set_variable = { trade_embargo_check^event_target:sph_emb_target = 1 }
										set_variable = { sph_forced_emb^event_target:sph_emb_target = 1 }
										PREV = { save_global_event_target_as = sph_news_actor }
										event_target:sph_emb_target = { save_global_event_target_as = sph_news_target }
										country_event = cwic_sphere.4
									}
									else_if = {
										limit = {
											check_variable = { sph_emb_on = 0 }
											check_variable = { sph_forced_emb^event_target:sph_emb_target = 1 }
										}
										set_variable = { trade_embargo_check^event_target:sph_emb_target = 0 }
										set_variable = { sph_forced_emb^event_target:sph_emb_target = 0 }
									}
								}
							}
						}
					}
				}
			}
		}
	}

	# ---- daily readout refresh (only the human's active targets) ----
	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				set_temp_variable = { sph_human_temp = THIS.id }
				sph_calc_slots = yes   # keep ROOT.sph_slots current for the readout
				# keep every Great Power's slot count current so selecting any of them shows it
				every_country = {
					limit = { is_sphere_capable = yes NOT = { tag = PREV } }
					sph_calc_slots = yes
				}
				for_each_loop = {
					array = sph_active_ids
					value = tid
					if = {
						limit = { country_exists = var:tid }
						var:tid = {
							set_variable = { sph_disp_active = 1 }
							set_temp_variable = { sph_from = sph_human_temp }
							sph_get_pair_index = yes
							if = {
								limit = { check_variable = { sph_idx > -1 } }
								set_variable = { sph_disp_score = sph_gp_score^sph_idx }
								set_variable = { sph_disp_stage = sph_gp_stage^sph_idx }
							}
							else = {
								set_variable = { sph_disp_score = 0 }
								set_variable = { sph_disp_stage = 0 }
							}
							# strongest rival track on this target (for the readout)
							set_variable = { sph_disp_rival_score = 0 }
							for_loop_effect = {
								end = sph_gp_ids^num
								value = v
								if = {
									limit = {
										NOT = { check_variable = { sph_gp_ids^v = sph_from } }
										check_variable = { sph_gp_score^v > sph_disp_rival_score }
									}
									set_variable = { sph_disp_rival_score = sph_gp_score^v }
								}
							}
						}
					}
				}
			}
		}
	}
}
